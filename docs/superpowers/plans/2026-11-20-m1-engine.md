# M1 — Engine CLI + Agent Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Chạy `autoedit edit --project <dir> --prompt "..."` từ terminal, ra `runs/N/final.mp4` đạt QC; chạy `autoedit analyze` ra 3 thẻ kịch bản; agent dùng subscription Claude qua `claude-agent-sdk`; engine đóng gói được bằng PyInstaller trên Mac.

**Architecture:** Python package `autoedit` (thư mục `engine/`). Prep (ffprobe/ffmpeg/whisper-cli) → agent (`claude-agent-sdk` `query()`, built-in tools tắt, chỉ có 5 tool MCP in-process + skill đóng gói trong `agent/`) → render bằng code theo Hard Rules của video-use → QC. Mọi tiến trình báo ra stdout dạng JSON-lines. Hàm `query` được inject để test không cần mạng.

**Tech Stack:** Python 3.12, uv, pytest, `claude-agent-sdk==0.2.186` (CLI Claude Code 2.1.329 nằm trong wheel, 353MB: không đóng gói, engine tải từ wheel PyPI ở lần chạy đầu), ffmpeg/ffprobe ≥ 7 (có libass, zscale), whisper.cpp `whisper-cli` + `ggml-large-v3-turbo.bin`, Pillow 13, PyInstaller 6.22.

**Spec:** `docs/superpowers/specs/2026-11-20-tiktok-autoedit-design.md` (mục 2, 3, 4, 5, 8, 9, 10)

## Global Constraints

- Clip đầu vào ≤ 180 giây; logo PNG/JPG; nhạc MP3/M4A/WAV. Sai → mã lỗi `BAD_INPUT`.
- Chuẩn hóa: 1080×1920, 30 fps CFR, SDR bt709; clip dọc → phóng lanczos kín khung + cắt giữa (không viền đen); clip ngang → cắt giữa + nền mờ trên/dưới.
- Cắt: bám ranh giới từ, đệm 50 ms trước / 80 ms sau, bỏ đoạn < 0.4 s, tổng giữ lại ≥ 30% thời lượng có lời. Chữ hook ≤ 5 từ.
- Render: cắt từng đoạn + afade 30 ms hai đầu, audio trung gian PCM, nối `-c copy`; overlay theo thứ tự hook → logo → CTA (2 s cuối) → **phụ đề cuối cùng**; loudnorm hai lượt -14 LUFS, TP -1 dB; libx264 CRF 18 cho bản cuối; preview 540p.
- QC: độ dài khớp EDL ±0.1 s; loudness -14±1 LUFS; không khung đen > 0.5 s.
- Agent: tối đa 15 lượt, 3 lần preview, timeout 10 phút; không có Bash/Read/Write/Edit; `RATE_LIMIT` không chạy dự phòng.
- Mã lỗi: `NO_AUTH`, `RATE_LIMIT`, `BAD_INPUT`, `RENDER_FAILED`, `AGENT_FAILED`, `SETUP_REQUIRED`.
- CLI Claude Code không nằm trong bộ cài: tải wheel `claude-agent-sdk==0.2.186` đúng nền tảng từ PyPI, kiểm sha256 ghim trong code, giải nén `claude`/`claude.exe` vào `<app data>/claude-cli/0.2.186/`. Mọi `ClaudeAgentOptions` truyền `cli_path` tới file đó (khi có) và `env={"DISABLE_AUTOUPDATER": "1"}`.
- Model mặc định `claude-opus-5-5`.
- Event stdout: `step{label,progress}`, `agent{text,tool}`, `preview{path,round}`, `done{final,summary,usage,needs_review}`, `error{code,message,retry_at}`. Mọi thứ khác ghi stderr.
- Không bao giờ ghi secret (OAuth token, API key) ra file hay log.
- Look theo ngành (spec 4.6): chỉ chỉnh luma bằng `eq` (`saturation=1.0`), trừ đồ ăn saturation 1.06; mịn da chỉ thời trang.
- Niche id: `thoi-trang`, `my-pham`, `do-an`, `phu-kien`, `chung`.

- [ ] **Step 3: Viết `probe_agent.py`**

```python
import asyncio, json, os, sys
from pathlib import Path
from claude_agent_sdk import (ClaudeAgentOptions, AssistantMessage, ResultMessage, RateLimitEvent,
                              SystemMessage, TextBlock, ToolUseBlock, create_sdk_mcp_server, query, tool)

BASE = Path(getattr(sys, "_MEIPASS", Path(__file__).parent))
calls = []

@tool("add", "Add two integers", {"a": int, "b": int})
async def add(args):
    calls.append(args)
    return {"content": [{"type": "text", "text": str(args["a"] + args["b"])}]}

async def main():
    server = create_sdk_mcp_server(name="spike", tools=[add])
    opts = ClaudeAgentOptions(
        cwd=str(BASE / "agent"),
        tools=["Skill"],
        skills=["hello"],
        setting_sources=["project"],
        mcp_servers={"spike": server},
        allowed_tools=["mcp__spike__add"],
        max_turns=6,
        model=os.environ.get("SPIKE_MODEL", "claude-opus-5-5"),
        cli_path=os.environ.get("SPIKE_CLI") or None,   # Step 6: CLI outside the package
        env={"DISABLE_AUTOUPDATER": "1"},
    )
    prompt = ("1) Use the add tool to compute 2+40. 2) Greet me in the house style. "
              "3) Try to run the shell command `ls` and tell me whether you could.")
    async for m in query(prompt=prompt, options=opts):
        if isinstance(m, SystemMessage) and m.subtype == "init":
            print("INIT tools:", m.data.get("tools"), "skills:", m.data.get("skills"), file=sys.stderr)
        elif isinstance(m, AssistantMessage):
            for b in m.content:
                if isinstance(b, ToolUseBlock):
                    print("TOOL_USE", b.name, json.dumps(b.input), file=sys.stderr)
                elif isinstance(b, TextBlock):
                    print("TEXT", b.text[:300], file=sys.stderr)
            if m.error:
                print("ASSISTANT_ERROR", m.error, file=sys.stderr)
        elif isinstance(m, RateLimitEvent):
            print("RATE_LIMIT", m.rate_limit_info, file=sys.stderr)
        elif isinstance(m, ResultMessage):
            print("RESULT", m.subtype, m.num_turns, m.is_error, (m.result or "")[:300], file=sys.stderr)
    print("CALLS", calls, file=sys.stderr)

asyncio.run(main())
```

import io, json
from autoedit.events import Emitter

def lines(buf):
    return [json.loads(l) for l in buf.getvalue().splitlines()]

def test_each_event_is_one_json_line():
    buf = io.StringIO()
    e = Emitter(buf)
    e.step("Đang nghe lời thoại…", 0.2)
    e.agent("Bỏ đoạn vấp", tool="set_edl")
    e.preview("/p/preview.mp4", 1)
    e.done("/p/final.mp4", ["a"], {"turns": 3}, False)
    e.error("RATE_LIMIT", "hết lượt", retry_at="2026-11-20T14:30:00+07:00")
    out = lines(buf)
    assert [o["type"] for o in out] == ["step", "agent", "preview", "done", "error"]
    assert out[0] == {"type": "step", "label": "Đang nghe lời thoại…", "progress": 0.2}
    assert out[1] == {"type": "agent", "text": "Bỏ đoạn vấp", "tool": "set_edl"}
    assert out[3]["needs_review"] is False
    assert out[4]["retry_at"].endswith("+07:00")

def test_non_ascii_is_kept_readable():
    buf = io.StringIO()
    Emitter(buf).step("Đang dựng", 0.5)
    assert "Đang dựng" in buf.getvalue()

def test_progress_is_clamped():
    buf = io.StringIO()
    Emitter(buf).step("x", 1.7)
    assert lines(buf)[0]["progress"] == 1.0
```

```python
"""JSON-lines events on stdout. The Tauri app reads one object per line."""
import json
import sys
from typing import TextIO


class Emitter:
    def __init__(self, stream: TextIO | None = None):
        self.stream = stream or sys.stdout

    def _emit(self, obj: dict) -> None:
        self.stream.write(json.dumps(obj, ensure_ascii=False) + "\n")
        self.stream.flush()

    def step(self, label: str, progress: float) -> None:
        self._emit({"type": "step", "label": label, "progress": round(min(1.0, max(0.0, progress)), 3)})

    def agent(self, text: str, tool: str | None = None) -> None:
        self._emit({"type": "agent", "text": text, "tool": tool})

    def preview(self, path: str, round: int) -> None:
        self._emit({"type": "preview", "path": path, "round": round})

    def done(self, final: str, summary: list[str], usage: dict, needs_review: bool) -> None:
        self._emit({"type": "done", "final": final, "summary": summary, "usage": usage,
                    "needs_review": needs_review})

    def error(self, code: str, message: str, retry_at: str | None = None) -> None:
        self._emit({"type": "error", "code": code, "message": message, "retry_at": retry_at})
```

class AutoeditError(Exception):
    code = "AGENT_FAILED"

    def __init__(self, message: str):
        super().__init__(message)
        self.message = message


class BadInput(AutoeditError):
    code = "BAD_INPUT"


class NoAuth(AutoeditError):
    code = "NO_AUTH"


class RenderFailed(AutoeditError):
    code = "RENDER_FAILED"


class AgentFailed(AutoeditError):
    code = "AGENT_FAILED"


class SetupRequired(AutoeditError):
    code = "SETUP_REQUIRED"


class RateLimited(AutoeditError):
    code = "RATE_LIMIT"

    def __init__(self, message: str, retry_at: str | None = None):
        super().__init__(message)
        self.retry_at = retry_at
```

"""Find sidecar binaries (bundled by Tauri next to the engine) and run them."""
import os
import shutil
import subprocess
import sys
from pathlib import Path

from autoedit.errors import RenderFailed

MODEL_NAME = "ggml-large-v3-turbo.bin"


def tool_path(name: str) -> str:
    exe = name + (".exe" if os.name == "nt" else "")
    if d := os.environ.get("AUTOEDIT_BIN_DIR"):
        p = Path(d) / exe
        if p.exists():
            return str(p)
    # Frozen: sidecars sit next to the engine executable.
    if getattr(sys, "frozen", False):
        p = Path(sys.executable).parent / exe
        if p.exists():
            return str(p)
    found = shutil.which(name)
    if not found:
        raise RenderFailed(f"Không tìm thấy công cụ {name}")
    return found


def app_data_dir() -> Path:
    if sys.platform == "darwin":
        return Path.home() / "Library" / "Application Support" / "AutoEdit"
    if os.name == "nt":
        return Path(os.environ.get("APPDATA", Path.home())) / "AutoEdit"
    return Path.home() / ".local" / "share" / "AutoEdit"


def model_path() -> Path:
    if env := os.environ.get("AUTOEDIT_MODEL"):
        return Path(env)
    return app_data_dir() / "models" / MODEL_NAME


def run(cmd: list[str], **kw) -> subprocess.CompletedProcess:
    proc = subprocess.run(cmd, capture_output=True, text=True, **kw)
    if proc.returncode != 0:
        tail = "\n".join(proc.stderr.strip().splitlines()[-20:])
        raise RenderFailed(f"{Path(cmd[0]).name} lỗi (mã {proc.returncode}):\n{tail}")
    return proc

```python
import json
from dataclasses import asdict, dataclass, field
from pathlib import Path

from autoedit.errors import BadInput

NICHES = ("thoi-trang", "my-pham", "do-an", "phu-kien", "chung")
_FIELDS = ("niche", "clip", "logo", "music", "prompt", "script_card", "runs")


@dataclass
class Project:
    dir: Path
    niche: str
    clip: str
    logo: str | None = None
    music: str | None = None
    prompt: str = ""
    script_card: dict | None = None
    runs: list[dict] = field(default_factory=list)

    @classmethod
    def load(cls, dir: Path) -> "Project":
        dir = Path(dir)
        f = dir / "project.json"
        if not f.exists():
            raise BadInput(f"Không thấy {f}")
        data = json.loads(f.read_text(encoding="utf-8"))
        p = cls(dir=dir, **{k: data[k] for k in _FIELDS if k in data})
        if p.niche not in NICHES:
            raise BadInput(f"Ngành không hợp lệ: {p.niche}")
        if not p.clip_path.exists():
            raise BadInput("Không thấy file clip trong project")
        p.work_dir.mkdir(parents=True, exist_ok=True)
        return p

    def save(self) -> None:
        data = {k: v for k, v in asdict(self).items() if k in _FIELDS}
        (self.dir / "project.json").write_text(json.dumps(data, ensure_ascii=False, indent=2), encoding="utf-8")

    @property
    def input_dir(self) -> Path:
        return self.dir / "input"

    @property
    def work_dir(self) -> Path:
        return self.dir / "work"

    @property
    def clip_path(self) -> Path:
        return self.input_dir / self.clip

    @property
    def logo_path(self) -> Path | None:
        return self.input_dir / self.logo if self.logo else None

    @property
    def music_path(self) -> Path | None:
        return self.input_dir / self.music if self.music else None

    def new_run_dir(self) -> Path:
        runs = self.dir / "runs"
        runs.mkdir(exist_ok=True)
        n = 1 + max((int(p.name) for p in runs.iterdir() if p.name.isdigit()), default=0)
        d = runs / str(n)
        d.mkdir()
        return d
```

```python
import json
from dataclasses import dataclass
from pathlib import Path

from autoedit import bins
from autoedit.errors import BadInput, RenderFailed

MAX_CLIP_SECONDS = 180
LOGO_EXT = {".png", ".jpg", ".jpeg"}
MUSIC_EXT = {".mp3", ".m4a", ".wav"}


@dataclass
class MediaInfo:
    duration: float
    width: int
    height: int
    fps: float
    hdr: bool
    has_audio: bool

    @property
    def portrait(self) -> bool:
        return self.height > self.width


def _fps(rate: str) -> float:
    num, _, den = rate.partition("/")
    return float(num) / float(den or 1) if float(den or 1) else 0.0


def probe(path: Path) -> MediaInfo:
    out = bins.run([bins.tool_path("ffprobe"), "-v", "error", "-print_format", "json",
                    "-show_format", "-show_streams", str(path)]).stdout
    data = json.loads(out)
    v = next((s for s in data.get("streams", []) if s.get("codec_type") == "video"), None)
    if v is None:
        raise BadInput("File không có hình")
    rotation = 0
    for sd in v.get("side_data_list", []) or []:
        if "rotation" in sd:
            rotation = int(sd["rotation"])
    w, h = int(v["width"]), int(v["height"])
    if abs(rotation) % 180 == 90:
        w, h = h, w
    return MediaInfo(
        duration=float(data["format"].get("duration") or v.get("duration") or 0),
        width=w, height=h,
        fps=_fps(v.get("avg_frame_rate") or v.get("r_frame_rate") or "30/1"),
        hdr=v.get("color_transfer") in ("smpte2084", "arib-std-b67"),
        has_audio=any(s.get("codec_type") == "audio" for s in data["streams"]),
    )


def check_clip(path: Path) -> MediaInfo:
    try:
        info = probe(path)
    except RenderFailed as e:
        raise BadInput("Không đọc được file clip") from e
    if info.duration > MAX_CLIP_SECONDS:
        raise BadInput(f"Clip dài {info.duration:.0f}s, tối đa {MAX_CLIP_SECONDS}s")
    if info.duration <= 0:
        raise BadInput("Clip không có thời lượng")
    return info


def check_logo(path: Path) -> None:
    if path.suffix.lower() not in LOGO_EXT or not path.exists():
        raise BadInput("Logo phải là PNG hoặc JPG")


def check_music(path: Path) -> None:
    if path.suffix.lower() not in MUSIC_EXT or not path.exists():
        raise BadInput("Nhạc phải là MP3, M4A hoặc WAV")
```

```python
from pathlib import Path

from autoedit import bins
from autoedit.prep.probe import MediaInfo

W, H, FPS = 1080, 1920, 30
TONEMAP = ("zscale=t=linear:npl=100,format=gbrpf32le,zscale=p=bt709,"
           "tonemap=tonemap=hable:desat=0,zscale=t=bt709:m=bt709:r=tv,format=yuv420p")


def build_filter(info: MediaInfo) -> str:
    pre = f"{TONEMAP}," if info.hdr else ""
    if info.portrait:
        # ffmpeg auto-rotates on decode. Fit inside 1080x1920 without upscaling, pad the rest.
        return (f"[0:v]{pre}scale=w='min({W},iw)':h='min({H},ih)':force_original_aspect_ratio=decrease,"
                f"pad={W}:{H}:(ow-iw)/2:(oh-ih)/2:color=black,setsar=1,fps={FPS},format=yuv420p[v]")
    # Landscape: sharp copy fitted to width, centred over a blurred, cropped fill.
    return (f"[0:v]{pre}split=2[bg][fg];"
            f"[bg]scale={W}:{H}:force_original_aspect_ratio=increase,crop={W}:{H},boxblur=20:2[bgb];"
            f"[fg]scale=w='min({W},iw)':h=-2[fgs];"
            f"[bgb][fgs]overlay=(W-w)/2:(H-h)/2,setsar=1,fps={FPS},format=yuv420p[v]")


def normalize(src: Path, dst: Path, info: MediaInfo) -> Path:
    dst.parent.mkdir(parents=True, exist_ok=True)
    cmd = [bins.tool_path("ffmpeg"), "-y", "-v", "error", "-i", str(src)]
    if not info.has_audio:
        cmd += ["-f", "lavfi", "-i", "anullsrc=r=48000:cl=stereo"]
    cmd += ["-filter_complex", build_filter(info), "-map", "[v]",
            "-map", "1:a" if not info.has_audio else "0:a:0",
            "-c:v", "libx264", "-preset", "fast", "-crf", "16",
            "-c:a", "aac", "-b:a", "192k", "-ar", "48000", "-ac", "2",
            "-t", f"{info.duration:.3f}", "-movflags", "+faststart", str(dst)]
    bins.run(cmd)
    return dst
```

from dataclasses import dataclass


@dataclass(frozen=True)
class Word:
    text: str
    start: float
    end: float


@dataclass(frozen=True)
class Phrase:
    start: float
    end: float
    text: str


def pack_phrases(words: list[Word], gap: float = 0.5) -> list[Phrase]:
    phrases: list[Phrase] = []
    cur: list[Word] = []
    for w in words:
        if cur and w.start - cur[-1].end >= gap:
            phrases.append(Phrase(cur[0].start, cur[-1].end, " ".join(x.text for x in cur)))
            cur = []
        cur.append(w)
    if cur:
        phrases.append(Phrase(cur[0].start, cur[-1].end, " ".join(x.text for x in cur)))
    return phrases


def format_phrases(phrases: list[Phrase]) -> str:
    return "\n".join(f"[{p.start:.2f}-{p.end:.2f}] {p.text}" for p in phrases)


def speech_duration(words: list[Word]) -> float:
    return sum(p.end - p.start for p in pack_phrases(words))


def words_to_json(words: list[Word]) -> list[dict]:
    return [{"text": w.text, "start": w.start, "end": w.end} for w in words]


def words_from_json(data: list[dict]) -> list[Word]:
    return [Word(d["text"], float(d["start"]), float(d["end"])) for d in data]

```python
import hashlib
import json
import tempfile
from pathlib import Path

from autoedit import bins
from autoedit.errors import RenderFailed
from autoedit.transcript import Word, words_from_json, words_to_json

WHISPER_LAG = 0.10


def parse_whisper_json(data: dict, lag: float = WHISPER_LAG) -> list[Word]:
    words = []
    for seg in data.get("transcription", []):
        text = (seg.get("text") or "").strip()
        if not text:
            continue
        off = seg["offsets"]
        start, end = off["from"] / 1000 + lag, off["to"] / 1000 + lag
        words.append(Word(text, round(start, 3), round(max(end, start), 3)))
    return words


def drop_hallucinations(words: list[Word]) -> list[Word]:
    """Whisper's invented lines get no acoustic alignment: runs of >=3 near-zero-length words."""
    out: list[Word] = []
    i = 0
    while i < len(words):
        j = i
        while j < len(words) and words[j].end - words[j].start < 0.05:
            j += 1
        if j - i >= 3:
            i = j
            continue
        out.extend(words[i:max(j, i + 1)])
        i = max(j, i + 1)
    return out


def _sha256(path: Path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for chunk in iter(lambda: f.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()


def _run_whisper(video: Path, work_dir: Path, language: str) -> list[Word]:
    model = bins.model_path()
    if not model.exists():
        raise RenderFailed("Chưa tải model nhận giọng nói")
    with tempfile.TemporaryDirectory() as tmp:
        wav = Path(tmp) / "a.wav"
        bins.run([bins.tool_path("ffmpeg"), "-y", "-v", "error", "-i", str(video),
                  "-ar", "16000", "-ac", "1", "-c:a", "pcm_s16le", str(wav)])
        base = Path(tmp) / "out"
        bins.run([bins.tool_path("whisper-cli"), "-m", str(model), "-f", str(wav), "-l", language,
                  "-ml", "1", "-sow", "-oj", "-of", str(base)])
        data = json.loads((base.with_suffix(".json")).read_text(encoding="utf-8"))
    return drop_hallucinations(parse_whisper_json(data))


def transcribe(video: Path, work_dir: Path, language: str = "vi") -> list[Word]:
    cache = work_dir / "transcript.json"
    digest = _sha256(video)
    if cache.exists():
        data = json.loads(cache.read_text(encoding="utf-8"))
        if data.get("sha256") == digest:
            return words_from_json(data["words"])
    words = _run_whisper(video, work_dir, language)
    work_dir.mkdir(parents=True, exist_ok=True)
    cache.write_text(json.dumps({"sha256": digest, "words": words_to_json(words)}, ensure_ascii=False),
                     encoding="utf-8")
    return words
```

from dataclasses import dataclass
from pathlib import Path

from autoedit.events import Emitter
from autoedit.prep.keyframes import extract_keyframes
from autoedit.prep.normalize import normalize
from autoedit.prep.probe import MediaInfo, check_clip, check_logo, check_music, probe
from autoedit.prep.transcribe import transcribe
from autoedit.project import Project
from autoedit.transcript import Word


@dataclass
class Prepared:
    info: MediaInfo
    normalized: Path
    words: list[Word]
    keyframes: list[Path]


def prepare(project: Project, events: Emitter) -> Prepared:
    events.step("Đang kiểm tra file…", 0.02)
    src_info = check_clip(project.clip_path)
    if project.logo_path:
        check_logo(project.logo_path)
    if project.music_path:
        check_music(project.music_path)

    norm = project.work_dir / "normalized.mp4"
    if not norm.exists() or norm.stat().st_mtime < project.clip_path.stat().st_mtime:
        events.step("Đang chuẩn hóa video…", 0.08)
        normalize(project.clip_path, norm, src_info)
    info = probe(norm)

    events.step("Đang nghe lời thoại…", 0.15)
    words = transcribe(norm, project.work_dir)

    frames_dir = project.work_dir / "frames"
    frames = sorted(frames_dir.glob("kf_*.jpg"))
    if len(frames) != 6:
        events.step("Đang xem các cảnh…", 0.22)
        frames = extract_keyframes(norm, frames_dir, info.duration, n=6)
    return Prepared(info=info, normalized=norm, words=words, keyframes=frames)

**Files:**
- Create: `engine/autoedit/edl.py`
- Test: `engine/tests/test_edl.py`

**Interfaces:**
- Consumes: `Word`, `speech_duration`
- Produces:
  - `@dataclass Range(start: float, end: float, reason: str = "")` + property `.duration`
  - `@dataclass Hook(start: float, end: float, text: str)`
  - `@dataclass CaptionFix(at: float, wrong: str, right: str)`
  - `@dataclass Emphasis(at: float, word: str)`
  - `@dataclass Edl(keep: list[Range], hook: Hook | None, caption_fixes: list[CaptionFix], emphasis: list[Emphasis], cta_text: str, music_use: bool, music_duck_db: float)` với `.to_dict() -> dict`, `.ordered() -> list[Range]` (hook trước, rồi các keep không trùng hook theo thời gian), `.output_duration() -> float`
  - `class EdlError(ValueError)` — message tiếng Việt/Anh ngắn gọn để trả lại agent
  - `validate_edl(raw: dict, words: list[Word], clip_duration: float) -> tuple[Edl, list[str]]` — trả EDL đã chỉnh + danh sách ghi chú (đã bám/đệm/bỏ gì). Ném `EdlError` khi vi phạm không tự sửa được.
  - Hằng số: `PAD_BEFORE = 0.05`, `PAD_AFTER = 0.08`, `MIN_SEGMENT = 0.4`, `MIN_KEEP_RATIO = 0.30`, `HOOK_MAX_WORDS = 5`, `DEFAULT_CTA = "BẤM GIỎ HÀNG 👇"`

Quy tắc của `validate_edl`:
1. `keep` bắt buộc, không rỗng; mỗi mục `start < end`, nằm trong `[0, clip_duration]`.
2. Bám mép: `start` → `start` của từ đầu tiên có `end > start` (nếu mép rơi giữa từ hoặc trong khoảng lặng trước từ đó); `end` → `end` của từ cuối có `start < end`. Đoạn không chứa từ nào (cảnh không lời) giữ nguyên mép.
3. Đệm: `start -= 0.05`, `end += 0.08`, kẹp trong `[0, clip_duration]` và không chồng lên đoạn kế tiếp (sau khi sắp xếp).
4. Gộp các đoạn chồng nhau; bỏ đoạn < 0.4 s (ghi chú).
5. Nếu transcript có lời: tổng thời lượng có lời nằm trong các đoạn giữ ≥ 30% `speech_duration` → nếu không, `EdlError`.
6. `hook`: `null` hoặc `{move_to_front: [a, b], text}`; `text` ≤ 5 từ (đếm theo khoảng trắng) → nếu không `EdlError`; khoảng `[a, b]` bám + đệm như trên.
7. `cta_text` rỗng → `DEFAULT_CTA`. `music.duck_db` kẹp trong `[-24, -6]`, mặc định -14.

```python
from dataclasses import asdict, dataclass, field

from autoedit.transcript import Word, speech_duration

PAD_BEFORE = 0.05
PAD_AFTER = 0.08
MIN_SEGMENT = 0.4
MIN_KEEP_RATIO = 0.30
HOOK_MAX_WORDS = 5
DEFAULT_CTA = "BẤM GIỎ HÀNG 👇"


class EdlError(ValueError):
    pass


@dataclass
class Range:
    start: float
    end: float
    reason: str = ""

    @property
    def duration(self) -> float:
        return self.end - self.start


@dataclass
class Hook:
    start: float
    end: float
    text: str


@dataclass
class CaptionFix:
    at: float
    wrong: str
    right: str


@dataclass
class Emphasis:
    at: float
    word: str


@dataclass
class Edl:
    keep: list[Range]
    hook: Hook | None = None
    caption_fixes: list[CaptionFix] = field(default_factory=list)
    emphasis: list[Emphasis] = field(default_factory=list)
    cta_text: str = DEFAULT_CTA
    music_use: bool = True
    music_duck_db: float = -14.0

    def to_dict(self) -> dict:
        return {
            "keep": [asdict(r) for r in self.keep],
            "hook": {"move_to_front": [self.hook.start, self.hook.end], "text": self.hook.text} if self.hook else None,
            "caption_fixes": [asdict(c) for c in self.caption_fixes],
            "emphasis": [asdict(e) for e in self.emphasis],
            "cta_text": self.cta_text,
            "music": {"use": self.music_use, "duck_db": self.music_duck_db},
        }

    def ordered(self) -> list[Range]:
        if not self.hook:
            return list(self.keep)
        h = Range(self.hook.start, self.hook.end, "hook")
        rest: list[Range] = []
        for r in self.keep:
            if r.end <= h.start or r.start >= h.end:
                rest.append(r)
                continue
            for a, b in ((r.start, h.start), (h.end, r.end)):
                if b - a >= MIN_SEGMENT:
                    rest.append(Range(a, b, r.reason))
        return [h] + rest

    def output_duration(self) -> float:
        return sum(r.duration for r in self.ordered())


def _snap(a: float, b: float, words: list[Word]) -> tuple[float, float]:
    inside = [w for w in words if w.end > a and w.start < b]
    if not inside:
        return a, b
    return inside[0].start, inside[-1].end


def _pad(a: float, b: float, dur: float) -> tuple[float, float]:
    return max(0.0, a - PAD_BEFORE), min(dur, b + PAD_AFTER)


def _speech_in(ranges: list[Range], words: list[Word]) -> float:
    total = 0.0
    for w in words:
        for r in ranges:
            total += max(0.0, min(w.end, r.end) - max(w.start, r.start))
    return total


def validate_edl(raw: dict, words: list[Word], clip_duration: float) -> tuple[Edl, list[str]]:
    notes: list[str] = []
    keep_raw = raw.get("keep") or []
    if not keep_raw:
        raise EdlError("keep must contain at least one range")
    ranges: list[Range] = []
    for k in keep_raw:
        a, b = float(k["start"]), float(k["end"])
        if not (0 <= a < b <= clip_duration + 0.01):
            raise EdlError(f"range {a}-{b} is outside 0-{clip_duration:.2f} or inverted")
        a, b = _pad(*_snap(a, min(b, clip_duration), words), clip_duration)
        ranges.append(Range(a, b, str(k.get("reason", ""))))

    ranges.sort(key=lambda r: r.start)
    merged: list[Range] = []
    for r in ranges:
        if merged and r.start <= merged[-1].end:
            merged[-1].end = max(merged[-1].end, r.end)
        else:
            merged.append(r)
    kept = []
    for r in merged:
        if r.duration < MIN_SEGMENT:
            notes.append(f"dropped {r.start:.2f}-{r.end:.2f}: shorter than {MIN_SEGMENT}s")
        else:
            kept.append(r)
    if not kept:
        raise EdlError("every range is shorter than 0.4s")

    total_speech = speech_duration(words)
    if total_speech > 0:
        ratio = _speech_in(kept, words) / total_speech
        if ratio < MIN_KEEP_RATIO:
            raise EdlError(f"keeps only {ratio:.0%} of the speech; keep at least 30%")

    hook = None
    if raw.get("hook"):
        h = raw["hook"]
        text = str(h.get("text", "")).strip()
        if len(text.split()) > HOOK_MAX_WORDS:
            raise EdlError(f"hook text must be at most {HOOK_MAX_WORDS} words")
        a, b = (float(x) for x in h["move_to_front"])
        if not (0 <= a < b <= clip_duration + 0.01):
            raise EdlError("hook range is outside the clip or inverted")
        a, b = _pad(*_snap(a, min(b, clip_duration), words), clip_duration)
        hook = Hook(a, b, text)

    music = raw.get("music") or {}
    duck = float(music.get("duck_db", -14.0))
    edl = Edl(
        keep=kept,
        hook=hook,
        caption_fixes=[CaptionFix(float(c["at"]), str(c["wrong"]), str(c["right"])) for c in raw.get("caption_fixes") or []],
        emphasis=[Emphasis(float(e["at"]), str(e["word"])) for e in raw.get("emphasis") or []],
        cta_text=(raw.get("cta_text") or "").strip() or DEFAULT_CTA,
        music_use=bool(music.get("use", True)),
        music_duck_db=min(-6.0, max(-24.0, duck)),
    )
    return edl, notes
```

"""Per-niche look. Luma-only `eq` keeps product/clothing colour true (edit-video lesson)."""

LOOKS: dict[str, dict] = {
    "thoi-trang": {"gamma": 1.08, "brightness": 0.015, "contrast": 1.02, "saturation": 1.0,
                   "skin": {"smooth": 6, "lift": 0.015}},
    "my-pham": {"gamma": 1.04, "brightness": 0.010, "contrast": 1.00, "saturation": 1.0, "skin": None},
    "do-an": {"gamma": 1.06, "brightness": 0.012, "contrast": 1.03, "saturation": 1.06, "skin": None},
    "phu-kien": {"gamma": 1.04, "brightness": 0.010, "contrast": 1.02, "saturation": 1.0, "skin": None},
    "chung": {"gamma": 1.05, "brightness": 0.010, "contrast": 1.00, "saturation": 1.0, "skin": None},
}


def look_graph(niche: str, inp: str, out: str, scale: str | None = None) -> str:
    lk = LOOKS[niche]
    eq = (f"eq=gamma={lk['gamma']}:brightness={lk['brightness']}:"
          f"contrast={lk['contrast']}:saturation={lk['saturation']}")
    tail = f"scale={scale},format=yuv420p" if scale else "format=yuv420p"
    skin = lk["skin"]
    if not skin:
        return f"[{inp}]{eq},{tail}[{out}]"
    # Chroma-box skin mask (edit-video): smooth + lift luma only inside the mask.
    return (
        f"[{inp}]format=yuv444p,{eq}[lk0];"
        f"[lk0]split=3[sk0][sk1][sk2];"
        f"[sk1]geq=lum='255*between(cb(X,Y),80,125)*between(cr(X,Y),135,170)*gt(lum(X,Y),75)':cb=128:cr=128,"
        f"format=gray,gblur=sigma=5,format=yuv444p[skm];"
        f"[sk2]bilateral=sigmaS={skin['smooth']}:sigmaR=0.06:planes=1,eq=brightness={skin['lift']}[sks];"
        f"[sk0][sks][skm]maskedmerge=planes=1,{tail}[{out}]"
    )
```

```python
"""Hard Rule 2 + 3: per-segment extract with 30ms audio fades, then lossless concat."""
from pathlib import Path

from autoedit import bins
from autoedit.edl import Range
from autoedit.render.look import look_graph

FADE = 0.03


def extract_segment(src: Path, rng: Range, niche: str, out: Path, preview: bool) -> Path:
    dur = rng.duration
    vgraph = look_graph(niche, "0:v", "v", scale="540:960" if preview else None)
    agraph = f"[0:a]afade=t=in:st=0:d={FADE},afade=t=out:st={max(0.0, dur - FADE):.3f}:d={FADE}[a]"
    preset, crf = ("veryfast", "24") if preview else ("fast", "16")
    bins.run([bins.tool_path("ffmpeg"), "-y", "-v", "error",
              "-ss", f"{rng.start:.3f}", "-i", str(src), "-t", f"{dur:.3f}",
              "-filter_complex", f"{vgraph};{agraph}", "-map", "[v]", "-map", "[a]",
              "-c:v", "libx264", "-preset", preset, "-crf", crf, "-pix_fmt", "yuv420p", "-r", "30",
              "-c:a", "pcm_s16le", "-ar", "48000", "-ac", "2", str(out)])
    return out


def render_base(src: Path, ranges: list[Range], niche: str, work: Path, preview: bool) -> Path:
    seg_dir = work / ("segs_preview" if preview else "segs")
    seg_dir.mkdir(parents=True, exist_ok=True)
    for old in seg_dir.glob("seg_*.mov"):
        old.unlink()
    segs = [extract_segment(src, r, niche, seg_dir / f"seg_{i:02d}.mov", preview) for i, r in enumerate(ranges)]
    lst = seg_dir / "concat.txt"
    lst.write_text("".join(f"file '{p.resolve().as_posix()}'\n" for p in segs), encoding="utf-8")
    base = work / ("base_preview.mov" if preview else "base.mov")
    bins.run([bins.tool_path("ffmpeg"), "-y", "-v", "error", "-f", "concat", "-safe", "0",
              "-i", str(lst), "-c", "copy", str(base)])
    return base
```

```python
"""Vietnamese captions on the OUTPUT timeline (Hard Rule 5), chunked per edit-video rules."""
import re
from dataclasses import dataclass
from pathlib import Path

from autoedit.edl import CaptionFix, Emphasis, Range
from autoedit.transcript import Word

FONT_NAME = "Be Vietnam Pro ExtraBold"
LEAD = {"các", "cực", "rất", "như", "một", "của", "cho", "mà", "có", "nhìn", "biết", "tất", "đến", "với",
        "là", "mọi", "nhà", "chiếc", "cái", "em", "những", "và", "thì", "để", "vì", "nên", "đã"}
GLUE = {"ơi", "yêu", "ứng", "kỳ", "ạ", "nha", "nhá", "í", "nhé", "luôn"}
PUNCT = ".,?!;:"


@dataclass
class TimedWord:
    text: str
    start: float
    end: float
    src_start: float
    group: int
    emph: bool = False


@dataclass
class Cue:
    start: float
    end: float
    words: list[TimedWord]


def _offsets(ranges: list[Range]) -> tuple[list[float], list[int]]:
    offs, off = [], 0.0
    for r in ranges:
        offs.append(off)
        off += r.duration
    groups = [0]
    for i in range(1, len(ranges)):
        jump = ranges[i].start - ranges[i - 1].end
        groups.append(groups[-1] + (1 if jump < 0 or jump > 2.0 else 0))
    return offs, groups


def _apply_fixes(words: list[Word], fixes: list[CaptionFix]) -> list[Word]:
    out = list(words)
    for f in fixes:
        n = len(f.wrong.split())
        cands = [i for i, w in enumerate(out) if abs(w.start - f.at) < 0.3
                 and w.text.casefold().strip(PUNCT) == f.wrong.split()[0].casefold()]
        if not cands:
            continue
        i = cands[0]
        last = out[min(i + n - 1, len(out) - 1)]
        out[i:i + n] = [Word(f.right, out[i].start, last.end)]
    return out


def map_words(ranges: list[Range], words: list[Word], fixes: list[CaptionFix],
              emphasis: list[Emphasis]) -> list[TimedWord]:
    offs, groups = _offsets(ranges)
    out: list[TimedWord] = []
    for w in _apply_fixes(words, fixes):
        best, best_ov = -1, 0.0
        for i, r in enumerate(ranges):
            ov = min(w.end, r.end) - max(w.start, r.start)
            if ov > best_ov:
                best, best_ov = i, ov
        if best < 0:
            continue
        r = ranges[best]
        emph = any(abs(e.at - w.start) < 0.3 and e.word.casefold() == w.text.casefold().strip(PUNCT)
                   for e in emphasis)
        out.append(TimedWord(w.text, max(r.start, w.start) - r.start + offs[best],
                             min(r.end, w.end) - r.start + offs[best], w.start, groups[best], emph))
    out.sort(key=lambda t: t.start)
    return out


def chunk(ws: list[TimedWord], target: int = 4, hard: int = 6) -> list[list[TimedWord]]:
    chunks: list[list[TimedWord]] = []
    cur: list[TimedWord] = []
    for i, w in enumerate(ws):
        nx = ws[i + 1] if i + 1 < len(ws) else None
        cur.append(w)
        if nx is None:
            break
        gap = nx.start - w.end
        core = w.text.rstrip(PUNCT).lower()
        if nx.text.rstrip(PUNCT).lower() in GLUE and len(cur) < hard:
            continue
        if ((w.text[-1] in PUNCT or gap >= 0.3) and len(cur) >= 2) or len(cur) >= hard:
            chunks.append(cur)
            cur = []
        elif len(cur) >= target and core not in LEAD and nx.text[-1] not in PUNCT:
            chunks.append(cur)
            cur = []
    if cur:
        chunks.append(cur)
    out: list[list[TimedWord]] = []
    for c in chunks:  # fold single-word orphans into the previous cue
        if len(c) == 1 and out and len(out[-1]) < hard:
            out[-1] = out[-1] + c
        else:
            out.append(c)
    return out


def build_cues(ranges: list[Range], words: list[Word], fixes: list[CaptionFix],
               emphasis: list[Emphasis], total: float) -> list[Cue]:
    tw = map_words(ranges, words, fixes, emphasis)
    groups: list[list[TimedWord]] = []
    for w in tw:
        if not groups or groups[-1][-1].group != w.group:
            groups.append([])
        groups[-1].append(w)
    cues = [Cue(c[0].start, max(c[-1].end, c[0].start + 0.3), c) for g in groups for c in chunk(g)]
    cues.sort(key=lambda c: c.start)
    for a, b in zip(cues, cues[1:]):
        a.end = min(a.end, b.start)
    if cues:
        cues[-1].end = max(cues[-1].end, total - 0.05)
    return cues


def speech_intervals(ranges: list[Range], words: list[Word], pad: float = 0.15) -> list[tuple[float, float]]:
    iv: list[list[float]] = []
    for w in map_words(ranges, words, [], []):
        a, b = max(0.0, w.start - pad), w.end + pad
        if iv and a - iv[-1][1] < 0.5:
            iv[-1][1] = max(iv[-1][1], b)
        else:
            iv.append([a, b])
    return [(a, b) for a, b in iv]


def _ts(t: float) -> str:
    cs = int(round(t * 100))
    h, rem = divmod(cs, 360000)
    m, rem = divmod(rem, 6000)
    s, cs = divmod(rem, 100)
    return f"{h}:{m:02d}:{s:02d}.{cs:02d}"


def write_ass(cues: list[Cue], path: Path) -> Path:
    head = (
        "[Script Info]\nScriptType: v4.00+\nPlayResX: 1080\nPlayResY: 1920\nWrapStyle: 0\n\n"
        "[V4+ Styles]\n"
        "Format: Name, Fontname, Fontsize, PrimaryColour, SecondaryColour, OutlineColour, BackColour, Bold, "
        "Italic, Underline, StrikeOut, ScaleX, ScaleY, Spacing, Angle, BorderStyle, Outline, Shadow, Alignment, "
        "MarginL, MarginR, MarginV, Encoding\n"
        f"Style: Caption,{FONT_NAME},78,&H00FFFFFF,&H00FFFFFF,&H00000000,&H00000000,0,0,0,0,100,100,0,0,1,5,0,2,"
        "90,150,540,1\n\n"
        "[Events]\nFormat: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text\n"
    )
    lines = []
    for c in cues:
        parts = []
        for w in c.words:
            t = re.sub(r"[{}\\]", "", w.text).rstrip(",;:.").upper()
            parts.append(f"{{\\c&H0000D7FF&}}{t}{{\\c&H00FFFFFF&}}" if w.emph else t)
        lines.append(f"Dialogue: 0,{_ts(c.start)},{_ts(c.end)},Caption,,0,0,0,,{' '.join(parts)}")
    path.write_text(head + "\n".join(lines) + "\n", encoding="utf-8")

"""Still overlays rendered with Pillow (CTA as PNG so emoji render, spec 4.5)."""
import os
import sys
from pathlib import Path

from PIL import Image, ImageDraw, ImageFont

from autoedit.resources import assets_dir

MAX_W = 960
PAD_X, PAD_Y = 36, 22


def font_path() -> Path:
    return assets_dir() / "fonts" / "BeVietnamPro-ExtraBold.ttf"


def emoji_font() -> tuple[Path, int] | None:
    if sys.platform == "darwin":
        p = Path("/System/Library/Fonts/Apple Color Emoji.ttc")
        return (p, 160) if p.exists() else None
    if os.name == "nt":
        p = Path(os.environ.get("WINDIR", r"C:\Windows")) / "Fonts" / "seguiemj.ttf"
        return (p, 109) if p.exists() else None
    return None


def _is_emoji(ch: str) -> bool:
    cp = ord(ch)
    return cp >= 0x1F000 or 0x2600 <= cp <= 0x27BF or cp in (0xFE0F, 0x200D)


def split_emoji(text: str) -> list[tuple[str, bool]]:
    runs: list[tuple[str, bool]] = []
    for ch in text:
        e = _is_emoji(ch)
        if runs and runs[-1][1] == e:
            runs[-1] = (runs[-1][0] + ch, e)
        else:
            runs.append((ch, e))
    return runs


def _render_line(line: str, font: ImageFont.FreeTypeFont, size: int) -> Image.Image:
    ef = emoji_font()
    tiles: list[Image.Image] = []
    for run, is_e in split_emoji(line):
        if is_e:
            if not ef:
                continue
            efont = ImageFont.truetype(str(ef[0]), ef[1])
            box = efont.getbbox(run)
            tile = Image.new("RGBA", (box[2] - box[0] + 4, box[3] - box[1] + 4), (0, 0, 0, 0))
            ImageDraw.Draw(tile).text((-box[0] + 2, -box[1] + 2), run, font=efont, embedded_color=True)
            h = int(size * 1.05)
            tiles.append(tile.resize((max(1, int(tile.width * h / tile.height)), h), Image.LANCZOS))
        else:
            box = font.getbbox(run)
            tile = Image.new("RGBA", (max(1, box[2]), int(size * 1.25)), (0, 0, 0, 0))
            ImageDraw.Draw(tile).text((0, int(size * 0.1)), run, font=font, fill=(255, 255, 255, 255))
            tiles.append(tile)
    w = sum(t.width for t in tiles) or 1
    h = max((t.height for t in tiles), default=int(size * 1.25))
    img = Image.new("RGBA", (w, h), (0, 0, 0, 0))
    x = 0
    for t in tiles:
        img.alpha_composite(t, (x, (h - t.height) // 2))
        x += t.width
    return img


def _wrap(text: str, font: ImageFont.FreeTypeFont) -> list[str]:
    lines, cur = [], ""
    for word in text.split():
        trial = f"{cur} {word}".strip()
        if cur and font.getlength(trial) > MAX_W - 2 * PAD_X:
            lines.append(cur)
            cur = word
        else:
            cur = trial
    return lines + ([cur] if cur else [])


def text_card(text: str, out: Path, size: int = 84, bg: tuple = (254, 44, 85, 235)) -> Path:
    font = ImageFont.truetype(str(font_path()), size)
    rendered = [_render_line(l, font, size) for l in _wrap(text.upper(), font)]
    w = min(MAX_W, max(r.width for r in rendered) + 2 * PAD_X)
    h = sum(r.height for r in rendered) + 2 * PAD_Y
    card = Image.new("RGBA", (w, h), (0, 0, 0, 0))
    ImageDraw.Draw(card).rounded_rectangle((0, 0, w - 1, h - 1), radius=28, fill=bg)
    y = PAD_Y
    for r in rendered:
        card.alpha_composite(r, ((w - r.width) // 2, y))
        y += r.height
    out.parent.mkdir(parents=True, exist_ok=True)
    card.save(out)
    return out


def prepare_logo(src: Path, out: Path, max_w: int = 200) -> Path:
    im = Image.open(src).convert("RGBA")
    h = max(1, round(im.height * max_w / im.width))
    im.resize((max_w, h), Image.LANCZOS).save(out)
    return out
```

import json
import subprocess
from pathlib import Path

from autoedit import bins

TARGET_I, TARGET_TP, TARGET_LRA = -14.0, -1.0, 11.0
_BASE = f"loudnorm=I={TARGET_I}:TP={TARGET_TP}:LRA={TARGET_LRA}"


def measure(path: Path) -> dict | None:
    proc = subprocess.run([bins.tool_path("ffmpeg"), "-hide_banner", "-nostats", "-i", str(path),
                           "-af", f"{_BASE}:print_format=json", "-vn", "-f", "null", "-"],
                          capture_output=True, text=True)
    s, e = proc.stderr.rfind("{"), proc.stderr.rfind("}")
    if s == -1 or e <= s:
        return None
    try:
        data = json.loads(proc.stderr[s:e + 1])
    except json.JSONDecodeError:
        return None
    need = {"input_i", "input_tp", "input_lra", "input_thresh", "target_offset"}
    return data if need <= data.keys() else None


def loudnorm(src: Path, dst: Path, two_pass: bool) -> Path:
    af = _BASE
    if two_pass and (m := measure(src)):
        af = (f"{_BASE}:measured_I={m['input_i']}:measured_TP={m['input_tp']}:measured_LRA={m['input_lra']}"
              f":measured_thresh={m['input_thresh']}:offset={m['target_offset']}:linear=true")
    bins.run([bins.tool_path("ffmpeg"), "-y", "-v", "error", "-i", str(src), "-c:v", "copy", "-af", af,
              "-c:a", "aac", "-b:a", "192k", "-ar", "48000", "-movflags", "+faststart", str(dst)])
    return dst

```python
"""Final pass: base -> hook -> logo -> CTA -> subtitles LAST (Hard Rule 1); music ducked under speech."""
from pathlib import Path, PurePath

from autoedit import bins
from autoedit.resources import assets_dir

HOOK_SECONDS = 2.5
CTA_SECONDS = 2.0


def filter_path(p: PurePath) -> str:
    return str(p).replace("\\", "/").replace(":", r"\:").replace("'", r"\'")


def music_volume_expr(intervals: list[tuple[float, float]], duck_db: float, base_db: float = -8.0) -> str:
    base = 10 ** (base_db / 20)
    duck = 10 ** (duck_db / 20)
    if not intervals:
        return f"{base:.3f}"
    speech = "+".join(f"between(t,{a:.3f},{b:.3f})" for a, b in intervals)
    return f"{base:.3f}*if(gt({speech},0),{duck:.3f},1)"


def compose(base: Path, out: Path, *, duration: float, ass: Path, hook_png: Path | None,
            logo_png: Path | None, cta_png: Path | None, music: Path | None,
            speech: list[tuple[float, float]], duck_db: float, preview: bool) -> Path:
    k = 0.5 if preview else 1.0
    cmd = [bins.tool_path("ffmpeg"), "-y", "-v", "error", "-i", str(base)]
    graph: list[str] = []
    cur, idx = "0:v", 1

    def add_overlay(png: Path, x: str, y: int, enable: str | None):
        nonlocal cur, idx
        cmd.extend(["-i", str(png)])
        src = f"{idx}:v"
        if preview:
            graph.append(f"[{src}]scale=iw*{k}:ih*{k}[ov{idx}]")
            src = f"ov{idx}"
        en = f":enable='{enable}'" if enable else ""
        graph.append(f"[{cur}][{src}]overlay=x={x}:y={int(y * k)}{en}[v{idx}]")
        cur, idx = f"v{idx}", idx + 1

    if hook_png:
        add_overlay(hook_png, "(W-w)/2", 300, f"between(t,0,{HOOK_SECONDS})")
    if logo_png:
        add_overlay(logo_png, str(int(48 * k)), 120, None)
    if cta_png:
        add_overlay(cta_png, "(W-w)/2", 1000, f"gte(t,{max(0.0, duration - CTA_SECONDS):.3f})")
    fonts = filter_path(assets_dir() / "fonts")
    graph.append(f"[{cur}]subtitles=filename='{filter_path(ass)}':fontsdir='{fonts}'"
                 f"{':original_size=1080x1920' if preview else ''}[vout]")

    if music:
        cmd.extend(["-stream_loop", "-1", "-i", str(music)])
        graph.append(f"[{idx}:a]aresample=48000,volume='{music_volume_expr(speech, duck_db)}':eval=frame,"
                     f"afade=t=out:st={max(0.0, duration - 1):.3f}:d=1[mus]")
        graph.append("[0:a][mus]amix=inputs=2:duration=first:normalize=0[aout]")
        amap = "[aout]"
    else:
        amap = "0:a"

    preset, crf = ("veryfast", "26") if preview else ("medium", "18")
    cmd += ["-filter_complex", ";".join(graph), "-map", "[vout]", "-map", amap,
            "-c:v", "libx264", "-preset", preset, "-crf", crf, "-pix_fmt", "yuv420p", "-r", "30",
            "-c:a", "aac", "-b:a", "192k", "-ar", "48000", "-t", f"{duration:.3f}",
            "-movflags", "+faststart", str(out)]
    bins.run(cmd)
    return out
```

import re
import subprocess
from dataclasses import dataclass, field
from pathlib import Path

from autoedit import bins
from autoedit.prep.probe import probe
from autoedit.render.loudness import TARGET_I, measure


@dataclass
class QcReport:
    ok: bool
    issues: list[str] = field(default_factory=list)
    duration: float = 0.0
    lufs: float | None = None


def run_qc(path: Path, expected_duration: float) -> QcReport:
    issues: list[str] = []
    dur = probe(path).duration
    if abs(dur - expected_duration) > 0.1:
        issues.append(f"thời lượng {dur:.2f}s khác dự kiến {expected_duration:.2f}s")
    m = measure(path)
    lufs = float(m["input_i"]) if m else None
    if lufs is None or abs(lufs - TARGET_I) > 1.0:
        issues.append(f"độ to {lufs} LUFS, cần {TARGET_I}±1")

```python
from dataclasses import dataclass
from pathlib import Path

from autoedit.edl import Edl
from autoedit.prep import Prepared
from autoedit.prep.keyframes import frame_at
from autoedit.project import Project
from autoedit.render.captions import build_cues, speech_intervals, write_ass
from autoedit.render.compose import compose
from autoedit.render.loudness import loudnorm
from autoedit.render.overlays import prepare_logo, text_card
from autoedit.render.qc import QcReport, run_qc
from autoedit.render.segments import render_base


@dataclass
class RenderResult:
    path: Path
    duration: float
    qc: QcReport | None
    needs_review: bool


def render(prepared: Prepared, project: Project, edl: Edl, out: Path, preview: bool) -> RenderResult:
    work = project.work_dir / ("render_preview" if preview else "render")
    work.mkdir(parents=True, exist_ok=True)
    ranges = edl.ordered()
    total = sum(r.duration for r in ranges)
    base = render_base(prepared.normalized, ranges, project.niche, work, preview)
    ass = write_ass(build_cues(ranges, prepared.words, edl.caption_fixes, edl.emphasis, total), work / "captions.ass")
    hook = text_card(edl.hook.text, work / "hook.png") if edl.hook and edl.hook.text else None
    logo = prepare_logo(project.logo_path, work / "logo.png") if project.logo_path else None
    cta = text_card(edl.cta_text, work / "cta.png", size=72, bg=(0, 0, 0, 200))
    music = project.music_path if (project.music_path and edl.music_use) else None
    speech = speech_intervals(ranges, prepared.words)

    def once() -> Path:
        mixed = compose(base, work / "composite.mp4", duration=total, ass=ass, hook_png=hook, logo_png=logo,
                        cta_png=cta, music=music, speech=speech, duck_db=edl.music_duck_db, preview=preview)
        return loudnorm(mixed, out, two_pass=not preview)

    once()
    if preview:
        return RenderResult(out, total, None, False)
    qc = run_qc(out, total)
    if not qc.ok:
        once()
        qc = run_qc(out, total)
    return RenderResult(out, total, qc, not qc.ok)


def preview_frames(video: Path, edl: Edl, out_dir: Path) -> list[Path]:
    times, t = [0.3], 0.0
    for r in edl.ordered()[:-1]:
        t += r.duration
        times.append(t + 0.15)
    return [frame_at(video, x, out_dir / f"pv_{i}.jpg", width=270) for i, x in enumerate(times[:4])]
```

# AutoEdit editor

You edit ONE short vertical sales clip (TikTok / Reels / Shorts) for a Vietnamese shop.
You work alone. There is no human to ask: never ask questions, pick the most obvious
interpretation and proceed. Everything you write for the customer is in Vietnamese.

## Tools (the only ones you have)

- `get_transcript(start?, end?)` – phrases with source times in seconds.
- `view_frames(times)` – up to 6 still frames at source times.
- `set_edl(edl)` – submit your edit. Code snaps every cut to word boundaries, pads it
  (50 ms before, 80 ms after), drops pieces under 0.4 s and rejects edits that keep under
  30% of the speech. It returns the corrected EDL or an error to fix.
- `render_preview()` – renders a 540p preview of the current EDL and returns frames at
  the start and at each cut. At most 3 previews.
- `finish(summary)` – locks the current EDL. The app renders the final video itself.

Budget: 15 tool calls in total. A good run uses 6–10.

## Process

1. Call `get_transcript()` once for the whole clip and read all of it.
2. Use the `niche-hooks` skill for the niche named in the request, and the
   `video-use` and `edit-video` skills for cut craft.
3. Call `view_frames` once with 3–6 times: the opening, product close-ups, the best visual moment.
4. Decide the hook, the order and the cuts. Call `set_edl`. If it returns an error, fix exactly
   that and call it again.
5. Call `render_preview` and check the frames. Re-submit with `set_edl` only if a cut is clearly
   wrong (mid-word, flash cut, hook not landing). Usually 1 preview is enough.
6. Call `finish` with 2–4 short customer-facing bullets.

## EDL

```json
{
  "keep": [{"start": 4.12, "end": 9.80, "reason": "take 2 tốt hơn take 1"}],
  "hook": {"move_to_front": [12.4, 15.1], "text": "CHÂN NGẮN? XEM NGAY"},
  "caption_fixes": [{"at": 18.2, "wrong": "tuyết mưu", "right": "tuyết mưa"}],
  "emphasis": [{"at": 21.0, "word": "189K"}],
  "cta_text": "BẤM GIỎ HÀNG 👇",
  "music": {"use": true, "duck_db": -14}
}
```

- `keep`: source ranges to keep, in source order. The video plays them in this order.
- `hook`: optional. A source range moved to the very start, plus an on-screen text of at most
  5 words shown for the first 2.5 s. Use `null` when nothing strong exists.
- `caption_fixes`: Whisper mistakes in brand names, prices, CTAs. `at` = source time of the first wrong word.
- `emphasis`: 1–4 words to highlight in captions (price, product name, key number).
- `cta_text`: shown for the last 2 s. Keep it short; an emoji is fine.
- `music.use`: false if the clip already has music or the speech is quiet.

## Hard rules

- Never cut inside a word. Never reorder words inside a sentence.
- Never invent claims or text. Hook text must be faithful to what is said.
- Forbidden promises, even if spoken: giảm cân, cao thêm X cm, trị mụn dứt điểm, trắng da sau N ngày,
  chữa bệnh, an toàn tuyệt đối. Cut those lines out.
- Never change product details, prices or colours. Keep the price line and the spoken CTA if present.
- Typical result: 15–45 s. Keep at least 30% of the speech.
- If the clip has no speech: choose the best visual moments with `view_frames`, `hook: null`.
- `finish` summary: plain Vietnamese, no jargon (no "EDL", "segment", "range").
  Example: "Bỏ 12 giây ngập ngừng ở đầu", "Đưa câu giá 189K lên mở đầu".
- If the request conflicts with these rules, follow the rules and say so in the summary.
```

---
name: video-use
description: Cut craft for talking and sales clips - choosing takes, where to cut, pacing, self-checking a preview. Use before writing any EDL.
---

Adapted from browser-use/video-use (MIT, see LICENSE). Rewritten for an agent that edits through
the `set_edl` / `render_preview` tools instead of running ffmpeg.

## Principles

1. Audio is primary, visuals follow. Cut candidates come from speech boundaries and silence gaps.
   Look at frames only at decision points (opening, product shots, ambiguous takes).
2. Reason from the raw transcript. Do not tag every filler up front; decide while cutting.
3. Verify your own output before finishing. If you would not ship it, fix it.

## Cut craft

- Cut in silences. Gaps of 0.3 s or more between words are safe cut points.
- Remove: long pauses, false starts, fillers (ừm, à, ờ, kiểu là), repeated sentences, off-topic asides.
- Retakes: when a line is said twice, keep the last complete, fluent take.
- Two neighbouring lines with no gap between them are one range. Do not split them.
- Prefer fewer, longer ranges over many tiny ones. Every range under 0.4 s is dropped anyway.
- Never cut inside a word. The tool snaps and pads edges, but choose edges in gaps anyway.

## Pacing for short vertical video

- First 3 seconds decide everything: open on the strongest line or visual (the hook).
- Remove dead air aggressively, but keep a breath before the price and before the CTA.
- End on the CTA line ("giỏ hàng", "đặt ngay", "inbox"). Nothing after it.

## Self-check on preview frames

- The first frame shows the product or the speaker clearly.
- No frame looks like a flash cut or a mid-gesture jump that breaks the sentence.
- The hook text fits the moment it covers.
- The last line is the CTA.
```

import base64
import json
from pathlib import Path

from claude_agent_sdk import create_sdk_mcp_server, tool

from autoedit import __version__
from autoedit.agent.session import EditSession
from autoedit.edl import EdlError, validate_edl
from autoedit.errors import AutoeditError
from autoedit.prep.keyframes import frame_at
from autoedit.render.pipeline import preview_frames, render
from autoedit.transcript import format_phrases, pack_phrases

MAX_PREVIEWS = 3
MAX_EDL_ERRORS = 3
MAX_FRAMES = 6
TOOL_NAMES = ("get_transcript", "view_frames", "set_edl", "render_preview", "finish")


def _text(t: str, error: bool = False) -> dict:
    res = {"content": [{"type": "text", "text": t}]}
    if error:
        res["is_error"] = True
    return res


def _image(p: Path) -> dict:
    return {"type": "image", "data": base64.b64encode(p.read_bytes()).decode(), "mimeType": "image/jpeg"}


async def get_transcript(s: EditSession, args: dict) -> dict:
    dur = s.prepared.info.duration
    a, b = float(args.get("start", 0.0)), float(args.get("end", dur))
    sel = [w for w in s.prepared.words if w.end > a and w.start < b]
    body = format_phrases(pack_phrases(sel)) or "(no speech in this range)"
    return _text(f"Clip duration {dur:.2f}s. Times are source seconds.\n{body}")


async def view_frames(s: EditSession, args: dict) -> dict:
    dur = s.prepared.info.duration
    times = [min(max(0.0, float(t)), max(0.0, dur - 0.05)) for t in args.get("times", [])][:MAX_FRAMES]
    if not times:
        return _text("times must contain 1-6 source times", error=True)
    out_dir = s.project.work_dir / "frames"
    content = [{"type": "text", "text": "Frames at " + ", ".join(f"{t:.2f}s" for t in times)}]
    for t in times:
        content.append(_image(frame_at(s.prepared.normalized, t, out_dir / f"view_{t:.2f}.jpg")))
    return {"content": content}


async def set_edl(s: EditSession, args: dict) -> dict:
    if s.edl_errors >= MAX_EDL_ERRORS and s.edl is None:
        return _text("Too many invalid edits. Stop and call finish.", error=True)
    try:
        edl, notes = validate_edl(args.get("edl") or {}, s.prepared.words, s.prepared.info.duration)
    except (EdlError, KeyError, TypeError, ValueError) as e:
        s.edl_errors += 1
        return _text(f"Invalid EDL: {e}", error=True)
    s.edl = edl
    (s.project.work_dir / "edl.json").write_text(json.dumps(edl.to_dict(), ensure_ascii=False, indent=2),
                                                 encoding="utf-8")
    msg = {"corrected_edl": edl.to_dict(), "notes": notes, "output_duration": round(edl.output_duration(), 2)}
    return _text(json.dumps(msg, ensure_ascii=False))


async def render_preview(s: EditSession, args: dict) -> dict:
    if s.edl is None:
        return _text("Call set_edl first.", error=True)
    if s.previews >= MAX_PREVIEWS:
        return _text("Preview limit reached. Call finish.", error=True)
    try:
        res = render(s.prepared, s.project, s.edl, s.project.work_dir / "preview.mp4", preview=True)
    except AutoeditError as e:
        return _text(f"Preview failed: {e.message}", error=True)
    s.previews += 1
    s.events.preview(str(res.path), s.previews)
    frames = preview_frames(res.path, s.edl, s.project.work_dir / "frames")
    content = [{"type": "text", "text": f"Preview {s.previews}/{MAX_PREVIEWS}: {res.duration:.2f}s. "
                                        "Frames: start, then just after each cut."}]
    return {"content": content + [_image(p) for p in frames]}


async def finish(s: EditSession, args: dict) -> dict:
    if s.edl is None:
        return _text("Call set_edl first.", error=True)
    summary = [str(x).strip() for x in args.get("summary", []) if str(x).strip()][:4]
    if not summary:
        return _text("summary must have 1-4 short Vietnamese bullets", error=True)
    s.summary, s.finished = summary, True
    return _text("Done. The app will render the final video.")


SCHEMAS = {
    "get_transcript": ("Read the transcript as phrases with source times. Optional start/end in seconds.",
                       {"type": "object", "properties": {"start": {"type": "number"}, "end": {"type": "number"}}}),
    "view_frames": ("See still frames at up to 6 source times (seconds).",
                    {"type": "object", "properties": {"times": {"type": "array", "items": {"type": "number"},
                                                                "minItems": 1, "maxItems": 6}},
                     "required": ["times"]}),
    "set_edl": ("Submit the edit decision list. Returns the corrected EDL or an error to fix.",
                {"type": "object", "properties": {"edl": {"type": "object"}}, "required": ["edl"]}),
    "render_preview": ("Render a 540p preview of the current EDL and return frames at the start and each cut.",
                       {"type": "object", "properties": {}}),
    "finish": ("Lock the current EDL. summary = 1-4 short customer-facing Vietnamese bullets.",
               {"type": "object", "properties": {"summary": {"type": "array", "items": {"type": "string"},
                                                             "minItems": 1, "maxItems": 4}},
                "required": ["summary"]}),
}
HANDLERS = {"get_transcript": get_transcript, "view_frames": view_frames, "set_edl": set_edl,
            "render_preview": render_preview, "finish": finish}


def build_server(session: EditSession):
    def bind(fn):
        async def handler(args):
            return await fn(session, args)
        return handler

    sdk_tools = [tool(name, SCHEMAS[name][0], SCHEMAS[name][1])(bind(HANDLERS[name])) for name in TOOL_NAMES]
    return create_sdk_mcp_server(name="autoedit", version=__version__, tools=sdk_tools)

```python
"""Download and locate the Claude Code CLI that matches the pinned claude-agent-sdk.

The CLI is not shipped in the installer: it is taken from the SDK's own PyPI wheel for this
platform, so its protocol always matches the SDK. Auto-update is disabled at call time
(see agent/options.py); upgrading means bumping SDK_VERSION + WHEELS and releasing the app.
"""
import hashlib
import os
import platform
import shutil
import stat
import sys
import urllib.request
import zipfile
from pathlib import Path

from autoedit.bins import app_data_dir
from autoedit.errors import SetupRequired
from autoedit.events import Emitter

SDK_VERSION = "0.2.186"
_PY = "https://files.pythonhosted.org/packages"
WHEELS = {
    ("darwin", "arm64"): (
        f"{_PY}/e4/23/c4a6dc2ae62899dcd15e5ec54e9da65fc62a6f88a00b67620aa8096109a8/"
        "claude_agent_sdk-0.2.186-py3-none-macosx_11_0_arm64.whl",
        "4f98e05618a955fc49190465b4eb01cb3c2fca6e0b9319ee6e6c6dd042619bc6"),
    ("darwin", "x86_64"): (
        f"{_PY}/aa/99/67caf906962bd7d3ecbe767cd058bb604bec10430a9eeba8c8a073f05a39/"
        "claude_agent_sdk-0.2.186-py3-none-macosx_11_0_x86_64.whl",
        "0099f4701569b7a625ef7a9c097577cf203671b5f793fec1697192d21c0be6f5"),
    ("win32", "AMD64"): (
        f"{_PY}/b0/bb/27cd01b7a9771d0db926fced99b2b88e7b35d26f6c832dd3aba5e7e1f6a9/"
        "claude_agent_sdk-0.2.186-py3-none-win_amd64.whl",
        "b68eefc87a966ec94d1d7d51ee97c01cd9849b8cef9da2e00066086a091c2716"),
}
CHUNK = 1 << 20


def platform_key() -> tuple[str, str]:
    return sys.platform, platform.machine()


def _exe() -> str:
    return "claude.exe" if sys.platform == "win32" else "claude"


def member_name() -> str:
    return f"claude_agent_sdk/_bundled/{_exe()}"


def cli_file() -> Path:
    return app_data_dir() / "claude-cli" / SDK_VERSION / _exe()


def find_cli() -> Path | None:
    if env := os.environ.get("AUTOEDIT_CLAUDE_CLI"):
        return Path(env)
    if cli_file().exists():
        return cli_file()
    if not getattr(sys, "frozen", False):
        return None
    raise SetupRequired("Chưa tải Claude. Mở app để tải lại.")


def install_cli(events: Emitter, *, opener=urllib.request.urlopen) -> Path:
    target = cli_file()
    if target.exists():
        return target
    try:
        url, sha = WHEELS[platform_key()]
    except KeyError:
        raise SetupRequired(f"Máy này chưa được hỗ trợ ({platform_key()[0]} {platform_key()[1]})") from None
    target.parent.mkdir(parents=True, exist_ok=True)
    wheel = target.parent / "sdk.whl.part"
    digest = hashlib.sha256()
    try:
        with opener(url) as resp, open(wheel, "wb") as f:
            total, done = int(resp.headers.get("Content-Length") or 0), 0
            while chunk := resp.read(CHUNK):
                f.write(chunk)
                digest.update(chunk)
                done += len(chunk)
                if total and done % (16 * CHUNK) < CHUNK:
                    events.step(f"Đang tải Claude… {done * 100 // total}%", done / total)
        if digest.hexdigest() != sha:
            raise SetupRequired("File Claude tải về bị lỗi, thử lại")
        tmp = target.with_name(target.name + ".part")
        with zipfile.ZipFile(wheel) as z, z.open(member_name()) as src, open(tmp, "wb") as dst:
            shutil.copyfileobj(src, dst, CHUNK)
        tmp.chmod(tmp.stat().st_mode | stat.S_IXUSR | stat.S_IXGRP | stat.S_IXOTH)
        os.replace(tmp, target)
    except (OSError, KeyError, zipfile.BadZipFile) as e:
        raise SetupRequired(f"Không tải được Claude: {e}") from e
    finally:
        for leftover in target.parent.glob("*.part"):
            leftover.unlink()
    for old in target.parent.parent.iterdir():
        if old.is_dir() and old.name != SDK_VERSION:
            shutil.rmtree(old, ignore_errors=True)
    events.step("Đã tải Claude", 1.0)
    return target
```

```python
import asyncio
import sys
from dataclasses import dataclass
from datetime import datetime

from claude_agent_sdk import AssistantMessage, RateLimitEvent, ResultMessage, TextBlock, ToolUseBlock, query

from autoedit.errors import NoAuth, RateLimited

TOOL_LABELS = {"get_transcript": "Đọc lời thoại", "view_frames": "Xem khung hình", "set_edl": "Chọn đoạn cắt",
               "render_preview": "Dựng bản xem nhanh", "finish": "Chốt bản dựng", "Skill": "Mở kiến thức dựng"}


@dataclass
class AgentOutcome:
    finished: bool
    turns: int
    error: str | None


def _iso(ts: int | None) -> str | None:
    return datetime.fromtimestamp(ts).astimezone().isoformat(timespec="seconds") if ts else None


async def run_agent(session, prompt: str, options, *, query_fn=query, timeout: float = 600) -> AgentOutcome:
    turns, reset_at = 0, None
    try:
        async with asyncio.timeout(timeout):
            async for msg in query_fn(prompt=prompt, options=options):
                if isinstance(msg, RateLimitEvent):
                    info = msg.rate_limit_info
                    reset_at = _iso(info.resets_at) or reset_at
                    if info.status == "rejected":
                        raise RateLimited("Đã hết lượt Claude của gói", reset_at)
                elif isinstance(msg, AssistantMessage):
                    if msg.error == "rate_limit":
                        raise RateLimited("Đã hết lượt Claude của gói", reset_at)
                    if msg.error in ("authentication_failed", "billing_error"):
                        raise NoAuth("Tài khoản Claude chưa đăng nhập hoặc không dùng được")
                    for b in msg.content:
                        if isinstance(b, ToolUseBlock):
                            name = b.name.removeprefix("mcp__autoedit__")
                            session.events.agent(TOOL_LABELS.get(name, name), tool=name)
                        elif isinstance(b, TextBlock) and b.text.strip():
                            session.events.agent(b.text.strip()[:280])
                elif isinstance(msg, ResultMessage):
                    turns = msg.num_turns
                    if msg.is_error:
                        print(f"agent result error: {msg.subtype} {msg.result}", file=sys.stderr)
    except TimeoutError:
        return AgentOutcome(False, turns, "timeout")
    return AgentOutcome(bool(session.finished and session.edl is not None), turns, None)
```

    return None

import base64
from pathlib import Path

from claude_agent_sdk import AssistantMessage, ResultMessage, query

from autoedit.agent.library import NICHE_NAMES, niche_section
from autoedit.agent.options import model, plain_options
from autoedit.errors import AgentFailed, NoAuth, RateLimited
from autoedit.prep import Prepared
from autoedit.transcript import format_phrases, pack_phrases

CARDS_SCHEMA = {
    "type": "object",
    "properties": {"cards": {"type": "array", "minItems": 3, "maxItems": 3, "items": {
        "type": "object",
        "properties": {"title": {"type": "string"}, "hook": {"type": "string"},
                       "outline": {"type": "array", "items": {"type": "string"}}, "prompt": {"type": "string"}},
        "required": ["title", "hook", "outline", "prompt"], "additionalProperties": False}}},
    "required": ["cards"], "additionalProperties": False,
}

INSTRUCTIONS = """You suggest 3 different edit scripts for a short vertical Vietnamese sales clip.
Each card: title (≤6 words), hook (on-screen text, ≤5 words, faithful to what is said),
outline (3-5 short steps in Vietnamese), prompt (one Vietnamese instruction an editor can follow,
mentioning which spoken line opens the video). Base every card on lines that exist in the transcript.
Use the niche library for structure and hook style. The 3 cards must differ in hook and order.
Never promise: giảm cân, cao thêm X cm, trị mụn dứt điểm, trắng da sau N ngày, chữa bệnh."""


def build_analyze_message(prepared: Prepared, niche: str, agent_dir: Path, exclude: list[str]) -> dict:
    transcript = format_phrases(pack_phrases(prepared.words)) or "(no speech - suggest visual-first scripts)"
    text = (f"{INSTRUCTIONS}\n\nNiche: {niche} ({NICHE_NAMES[niche]})\n"
            f"Niche library:\n{niche_section(niche, agent_dir)}\n\n"
            f"Clip duration: {prepared.info.duration:.1f}s\nTranscript:\n{transcript}\n")
    if exclude:
        text += "\nDo not repeat these earlier cards: " + "; ".join(exclude) + "\n"
    content = [{"type": "text", "text": text}]
    for kf in prepared.keyframes[:3]:
        content.append({"type": "image", "source": {"type": "base64", "media_type": "image/jpeg",
                                                    "data": base64.b64encode(kf.read_bytes()).decode()}})
    return {"type": "user", "message": {"role": "user", "content": content},
            "parent_tool_use_id": None, "session_id": "default"}


async def run_analyze(prepared: Prepared, niche: str, agent_dir: Path, exclude: list[str] | None = None,
                      *, query_fn=query) -> list[dict]:
    message = build_analyze_message(prepared, niche, agent_dir, exclude or [])

    async def stream():
        yield message

    opts = plain_options(agent_dir, model(), max_turns=3,
                         output_format={"type": "json_schema", "schema": CARDS_SCHEMA})
    structured = None
    async for msg in query_fn(prompt=stream(), options=opts):
        if isinstance(msg, AssistantMessage):
            if msg.error == "rate_limit":
                raise RateLimited("Đã hết lượt Claude của gói")
            if msg.error in ("authentication_failed", "billing_error"):
                raise NoAuth("Tài khoản Claude không dùng được")
        elif isinstance(msg, ResultMessage):
            structured = msg.structured_output
    cards = (structured or {}).get("cards") if isinstance(structured, dict) else None
    if not cards or len(cards) != 3 or not all({"title", "hook", "outline", "prompt"} <= set(c) for c in cards):
        raise AgentFailed("AI không gợi ý được kịch bản, thử lại")
    return cards
```

import argparse
import asyncio
import json
import shutil
import sys
import traceback
from pathlib import Path

from autoedit import __version__
from autoedit.agent.analyze import run_analyze
from autoedit.agent.auth import auth_check
from autoedit.agent.options import edit_options, model
from autoedit.agent.prompt import build_edit_prompt
from autoedit.agent.runner import run_agent
from autoedit.agent.session import EditSession
from autoedit.agent.tools import build_server
from autoedit.claude_cli import find_cli, install_cli
from autoedit.errors import AutoeditError, NoAuth, RateLimited
from autoedit.events import Emitter
from autoedit.fallback import FALLBACK_SUMMARY, silence_cut_edl
from autoedit.prep import prepare
from autoedit.project import Project
from autoedit.render.pipeline import render
from autoedit.resources import agent_dir


class Tee:
    def __init__(self, *streams):
        self.streams = streams

    def write(self, s: str) -> None:
        for st in self.streams:
            st.write(s)

    def flush(self) -> None:
        for st in self.streams:
            st.flush()


def _previous_edl(project: Project) -> dict | None:
    for run in reversed(project.runs):
        f = project.dir / "runs" / str(run["n"]) / "edl.json"
        if f.exists():
            return json.loads(f.read_text(encoding="utf-8"))
    return None


async def _edit(args, events: Emitter) -> int:
    find_cli()  # fail fast with SETUP_REQUIRED, before transcribing
    project = Project.load(Path(args.project))
    prepared = prepare(project, events)
    user_prompt = args.prompt if args.prompt is not None else project.prompt
    previous = _previous_edl(project) if args.feedback else None
    run_dir = project.new_run_dir()
    with open(run_dir / "agent.jsonl", "a", encoding="utf-8") as log:
        ev = Emitter(Tee(events.stream, log))
        session = EditSession(prepared=prepared, project=project, events=ev)
        usage = {"turns": 0, "fallback": False}
        ev.step("AI đang dựng…", 0.3)
        try:
            outcome = await run_agent(session, build_edit_prompt(project, prepared, user_prompt, args.feedback, previous),
                                      edit_options(agent_dir(), build_server(session), model()))
            usage["turns"] = outcome.turns
            ok = outcome.finished
        except (RateLimited, NoAuth):
            log.close()
            shutil.rmtree(run_dir, ignore_errors=True)
            raise
        except Exception:
            traceback.print_exc(file=sys.stderr)
            ok = False
        if ok:
            edl, summary = session.edl, session.summary
        else:
            edl, summary = silence_cut_edl(prepared.words, prepared.info.duration), [FALLBACK_SUMMARY]
            usage["fallback"] = True
        ev.step("Đang xuất video…", 0.9)
        res = render(prepared, project, edl, run_dir / "final.mp4", preview=False)
        (run_dir / "edl.json").write_text(json.dumps(edl.to_dict(), ensure_ascii=False, indent=2), encoding="utf-8")
        project.prompt = user_prompt
        project.runs.append({"n": int(run_dir.name), "final": f"runs/{run_dir.name}/final.mp4",
                             "prompt": user_prompt, "feedback": args.feedback, "summary": summary,
                             "needs_review": res.needs_review, "fallback": usage["fallback"]})
        project.save()
        ev.done(str(res.path), summary, usage, res.needs_review)
    return 0


async def _analyze(args, events: Emitter) -> int:
    find_cli()
    project = Project.load(Path(args.project))
    prepared = prepare(project, events)
    events.step("AI đang gợi ý kịch bản…", 0.6)
    events.cards(await run_analyze(prepared, project.niche, agent_dir(), args.exclude or []))
    return 0


async def _setup(args, events: Emitter) -> int:
    events.setup(str(install_cli(events)))
    return 0


async def _auth(args, events: Emitter) -> int:
    events.auth(await auth_check(agent_dir()))
    return 0


def main(argv: list[str] | None = None) -> int:
    p = argparse.ArgumentParser(prog="autoedit")
    p.add_argument("--version", action="store_true")
    sub = p.add_subparsers(dest="cmd")
    a = sub.add_parser("analyze"); a.add_argument("--project", required=True); a.add_argument("--exclude", action="append")
    e = sub.add_parser("edit"); e.add_argument("--project", required=True); e.add_argument("--prompt"); e.add_argument("--feedback")
    sub.add_parser("auth-check")
    sub.add_parser("setup")
    args = p.parse_args(argv)
    if args.version:
        print(__version__)
        return 0
    events = Emitter()
    handler = {"analyze": _analyze, "edit": _edit, "auth-check": _auth, "setup": _setup}.get(args.cmd)
    if handler is None:
        p.print_help(sys.stderr)
        return 1
    try:
        return asyncio.run(handler(args, events))
    except RateLimited as err:
        events.error(err.code, err.message, err.retry_at)
        return 2
    except AutoeditError as err:
        events.error(err.code, err.message)
        return 1
    except Exception as err:  # never leave the app without a terminal event
        traceback.print_exc(file=sys.stderr)
        events.error("AGENT_FAILED", str(err))
        return 1
```

```python
"""Run every clip in bench/clips through `autoedit edit` and write bench/out/report.html."""
import html
import json
import shutil
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).parent
CLIPS, OUT = ROOT / "clips", ROOT / "out"


def _thumbs(video: Path, out_dir: Path, duration: float) -> list[str]:
    paths = []
    for i, t in enumerate((0.5, duration / 2, max(0.0, duration - 1))):
        p = out_dir / f"t{i}.jpg"
        subprocess.run(["ffmpeg", "-y", "-v", "error", "-ss", f"{t:.2f}", "-i", str(video), "-frames:v", "1",
                        "-vf", "scale=180:-2", str(p)], check=False)
        if p.exists():
            paths.append(p.relative_to(OUT).as_posix())
    return paths


def run_one(src: Path) -> dict:
    name, niche = src.name, src.name.rsplit("-", 1)[0]
    proj = OUT / name
    shutil.rmtree(proj, ignore_errors=True)
    (proj / "input").mkdir(parents=True)
    data = {"niche": niche, "clip": "clip.mp4"}
    shutil.copy(src / "clip.mp4", proj / "input" / "clip.mp4")
    for key, fname in (("logo", "logo.png"), ("music", "music.mp3")):
        if (src / fname).exists():
            shutil.copy(src / fname, proj / "input" / fname)
            data[key] = fname
    (proj / "project.json").write_text(json.dumps(data), encoding="utf-8")
    prompt = (src / "prompt.txt").read_text(encoding="utf-8").strip() if (src / "prompt.txt").exists() else ""
    proc = subprocess.run([sys.executable, "-m", "autoedit", "edit", "--project", str(proj), "--prompt", prompt],
                          capture_output=True, text=True)
    events = [json.loads(l) for l in proc.stdout.splitlines() if l.startswith("{")]
    last = events[-1] if events else {"type": "error", "code": "NO_OUTPUT", "message": proc.stderr[-500:]}
    if last["type"] != "done":
        return {"name": name, "niche": niche, "status": "error", "error": last.get("code"), "thumbs": []}
    final = Path(last["final"])
    dur = float(subprocess.run(["ffprobe", "-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0",
                                str(final)], capture_output=True, text=True).stdout or 0)
    return {"name": name, "niche": niche, "status": "done", "duration": round(dur, 1),
            "needs_review": last["needs_review"], "fallback": last["usage"].get("fallback", False),
            "summary": last["summary"], "thumbs": _thumbs(final, proj, dur)}


def build_report(results: list[dict], out: Path) -> Path:
    rows = ["<tr><th>Clip</th><th>Ngành</th><th>Kết quả</th><th>Thời lượng</th><th>QC</th><th>Dự phòng</th>"
            "<th>AI đã làm</th><th>Ảnh</th></tr>"]
    for r in results:
        ok = r["status"] == "done"
        imgs = "".join(f'<img src="{html.escape(t)}" height="160">' for t in r.get("thumbs", []))
        rows.append(
            f"<tr><td>{html.escape(r['name'])}</td><td>{r['niche']}</td>"
            f"<td>{'xong' if ok else html.escape(str(r.get('error')))}</td>"
            f"<td>{r.get('duration', '')}</td><td>{'⚠️' if r.get('needs_review') else ('✓' if ok else '')}</td>"
            f"<td>{'có' if r.get('fallback') else ''}</td>"
            f"<td>{'<br>'.join(html.escape(s) for s in r.get('summary', []))}</td><td>{imgs}</td></tr>")
    out.write_text("<meta charset='utf-8'><style>td,th{border:1px solid #ccc;padding:6px;vertical-align:top}"
                   "table{border-collapse:collapse;font:13px sans-serif}</style><table>" + "".join(rows) + "</table>",
                   encoding="utf-8")
    return out


def main() -> None:
    OUT.mkdir(exist_ok=True)
    clips = sorted(p for p in CLIPS.iterdir() if (p / "clip.mp4").exists())
    results = []
    for c in clips:
        print(f"→ {c.name}", flush=True)
        results.append(run_one(c))
    print(build_report(results, OUT / "report.html"))


if __name__ == "__main__":
    main()
```
